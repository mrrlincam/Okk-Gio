app.post("/api/download", async (req, res) => {

    try {

        const { url, quality } = req.body;

        if (!url) {

            return res.status(400).json({
                error: "TikTok URL required"
            });

        }

        const response = await axios.post(

            "https://www.tikwm.com/api/",

            new URLSearchParams({
                url: url,
                hd: "1"
            }),

            {
                headers: {
                    "Content-Type":
                    "application/x-www-form-urlencoded",

                    "User-Agent":
                    "Mozilla/5.0"
                },

                timeout: 30000

            }

        );

        const data = response.data.data;

        if (
            response.data.code !== 0 ||
            !data
        ) {

            throw new Error("Unable to get TikTok video");

        }

        let videoUrl;

        // Select the best available source.
        // TikWM does not guarantee exact resolutions.

        if (quality === "720p") {

            videoUrl = data.play || data.hdplay;

        } else {

            videoUrl = data.hdplay || data.play;

        }

        if (!videoUrl) {

            throw new Error("Video URL unavailable");

        }

        const token = crypto.randomBytes(16).toString("hex");

        downloads.set(token, {

            url: videoUrl,

            title: data.title || "TikTok Video",

            created: Date.now()

        });

        res.json({

            success: true,

            title: data.title || "TikTok Video",

            cover: data.cover || "",

            author:
                data.author?.nickname || "TikTok Creator",

            quality: quality || "1080p",

            downloadUrl: "/api/file/" + token

        });

    } catch (error) {

        console.error(error.message);

        res.status(500).json({

            error: "Cannot process video. Please try again."

        });

    }

});

app.get("/api/file/:token", async (req, res) => {

    try {

        const item = downloads.get(req.params.token);

        if (!item) {

            return res.status(404).send(
                "Download expired. Please process the video again."
            );

        }

        // Token expires after 30 minutes.

        if (Date.now() - item.created > 30 * 60 * 1000) {

            downloads.delete(req.params.token);

            return res.status(410).send(
                "Download link expired."
            );

        }

        const video = await axios.get(item.url, {

            responseType: "stream",

            timeout: 60000,

            headers: {

                "User-Agent": "Mozilla/5.0"

            }

        });

        res.setHeader(

            "Content-Type",

            video.headers["content-type"] ||
            "video/mp4"

        );

        res.setHeader(

            "Content-Disposition",

            'attachment; filename="ZHEA-GIO-VIDEO.mp4"'

        );

        video.data.pipe(res);

        video.data.on("error", () => {

            if (!res.headersSent) {
                res.status(500);
            }

            res.end();

        });

    } catch (error) {

        console.error(error.message);

        if (!res.headersSent) {

            res.status(500).send(
                "Video download failed."
            );

        }

    }

});
